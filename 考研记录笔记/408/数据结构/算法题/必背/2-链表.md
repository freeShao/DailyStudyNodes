# 单链表（带头结点）

```cpp
typedef int ElemType;
typedef struct LNode {
    ElemType data; // 亦可直接写 int data;
    struct LNode *next;
} LNode, *LinkList; // 有时题目自定义为[node, *link]，更换表达即可

// 一般使用方法：
void func(LinkList L){ // L 代指头结点
    P = (LNode*) malloc (sizeof(LNode)); // 创建头
    P->next = NULL;
    LNode *p = L->next; // 首个结点
    LNode *q; // 单个结点
    free(q); // 释放p
    return;
}
```

```cpp
void HeadInsert(LinkList &L, int a[], int n) { // [*] 此处使用&L，将原链表重构，指向新的头结点
    L = (LNode *)malloc(sizeof(LNode)); // 创建头结点
    L->next = NULL;
    
    LNode* s; // 无论是 [LNode* s/LNode *s] 均可

    int x;
```

## 4. 双指针(常用) [一般是快慢+同步]

```cpp
void func(LinkList L, int n, int k) { // 通常来说k小于链表长n，需要额外边界判定[也可后续判定]
    // 当链表为空或者快指针越界
    if (L == NULL && L->next == NULL || n<k) {return;}
    // 当链表长度未知时，需要结合 [1.求表长]

    LNode *fast = L->next; // 定义快慢指针
    LNode *slow = L->next;  // 初始化为链表首个结点 

    // 1. 快指针先走for循环
    for (int i; i<k; i++) { // 快指针先走k步
        fast = fast->next;
    }
    // 或者以下方式
    // 1.1. 快指针先走while循环兼顾边界判定
    int i=0;
    while (fast != NULL && i<k) {
        fast = fast->next;
        i++;
    }
    if (i<k) return; // 注意是void还是int，如果是int，需要返回0

    // 2. 指针同步移动
    while (fast != NULL) {
        fast = fast->next;
        slow = slow->next;
        // if ...{...; return;} 满足某条件时可提前跳出
    }
    // 或者以下方式
    // 2.1. 指针同步移动
    while (fast->next != NULL && fast->next->next != NULL) { // 找链表终点
        slow = slow->next; // 快指针走两步，满指针走一步
        fast = fast->next->next;
    }
    printf("%d", slow->data);
    return;
}
```

## 5. 链表原地逆置

```cpp
void ReverseList(LinkList &L) {
    // 若链表为空或只有一个结点时无须逆置
    if (L == NULL || L->next == NULL || L->next->next == NULL) return;

    LNode *prev = NULL; // 末位
    LNode *curr = L->next; // 首个结点
    LNode *next; // 下一位
    while (curr != NULL) { // 下一位置非空时(遍历链表)
        next = curr->next; // 保存当前结点的下一结点
        curr->next = prev; // 当前结点指向前一个结点(正式逆置)
        prev = next; // 前一个结点指针后移
        curr = next; // 当前结点指针后移
    }
    L->next = prev; // 头结点指向逆置后最后一个结点
}
```
